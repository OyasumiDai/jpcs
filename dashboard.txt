    $event_id = $_POST['event_id'];

    if (!is_dir($targetDir)) mkdir($targetDir, 0777, true);

    foreach ($_FILES['images']['name'] as $key => $name) {
        $filePath = $targetDir . time() . "_" . basename($name);
        move_uploaded_file($_FILES['images']['tmp_name'][$key], $filePath);
        $conn->query("INSERT INTO highlight_images (event_id, image_path, uploaded_at) 
                      VALUES ('$event_id','$filePath', NOW())");
    }

    header("Location: dashboard.php?success=" . urlencode("Images added to highlight!"));
    exit();

  <script>
    let currentImages = [];
    let currentIndex = 0;
    function openViewer(images, index) {
      currentImages = images;
      currentIndex = index;
      document.getElementById('viewerImg').src = currentImages[currentIndex].image_path;
      document.getElementById('viewerModal').style.display = 'flex';
    }
    function closeViewer() { document.getElementById('viewerModal').style.display = 'none'; }
    function nextImage() {
      if (currentIndex < currentImages.length - 1) {
        currentIndex++;
        document.getElementById('viewerImg').src = currentImages[currentIndex].image_path;
      }
    }
    function prevImage() {
      if (currentIndex > 0) {
        currentIndex--;
        document.getElementById('viewerImg').src = currentImages[currentIndex].image_path;
      }
    }
